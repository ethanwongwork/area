.rt-Inset {
  box-sizing: border-box;

  /* We reset the defined margin variables to avoid inheriting them from a higher component */
  /* If a margin IS defined on the component itself, the utility class will win and reset it */
  --margin-top: 0px;
  --margin-right: 0px;
  --margin-bottom: 0px;
  --margin-left: 0px;

  overflow: hidden;

  /*
   * Note: Next.js may collapse this into a shorthand "margin" declaration when bundling the CSS.
   * If any of the variables are set to "initial", the shorthand syntax will become invalid and botch the Inset styles.
   */
  margin-top: var(--margin-top-override);   /* = initial */
  margin-right: var(--margin-right-override);   /* = initial */
  margin-bottom: var(--margin-bottom-override);   /* = initial */
  margin-left: var(--margin-left-override);   /* = initial */

  /* Reset the overrides on direct children */
  :where(&) > * {
    --margin-top-override: initial;
    --margin-right-override: initial;
    --margin-bottom-override: initial;
    --margin-left-override: initial;
  }
}

    /* We reset the defined margin variables to avoid inheriting them from a higher component */

    --margin-top: 0px;
    --margin-right: 0px;
    --margin-bottom: 0px;
    --margin-left: 0px;

    /* Define the overrides to incorporate the negative margins */
    --margin-top-override: calc(var(--margin-top) - var(--button-ghost-padding-y));   /* = -4px */
    --margin-right-override: calc(var(--margin-right) - var(--button-ghost-padding-x));   /* = -8px */
    --margin-bottom-override: calc(var(--margin-bottom) - var(--button-ghost-padding-y));   /* = -4px */
    --margin-left-override: calc(var(--margin-left) - var(--button-ghost-padding-x));   /* = -8px */

    /* Reset the overrides on direct children */
    :where(&) > * {
      --margin-top-override: initial;
      --margin-right-override: initial;
      --margin-bottom-override: initial;
      --margin-left-override: initial;
    }

    margin: var(--margin-top-override) var(--margin-right-override) var(--margin-bottom-override)   /* = -4px -8px -4px */
      var(--margin-left-override);
  }
}

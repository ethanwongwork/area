/* surface */

.rt-BaseCheckboxRoot:where(.rt-variant-surface) {
  &:where([data-state='unchecked']) {
    &::before {
      background-color: var(--color-surface);   /* = rgba(255, 255, 255, 0.85) */
      box-shadow: inset 0 0 0 1px var(--gray-a7);   /* = inset 0 0 0 1px var(--mauve-a7) */
    }
  }

  &:where([data-state='checked'], [data-state='indeterminate']) {
    &::before {
      background-color: var(--accent-indicator);   /* = var(--amber-indicator) */

      }
    }
  }

    }
  }
}

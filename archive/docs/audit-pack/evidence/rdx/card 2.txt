.rt-Card:where(.rt-variant-ghost) {
  --card-border-width: 0px;

  /* We reset the defined margin variables to avoid inheriting them from a higher component */
  /* If a margin IS defined on the component itself, the utility class will win and reset it */
  --margin-top: 0px;
  --margin-right: 0px;
  --margin-bottom: 0px;
  --margin-left: 0px;

  /* Define the overrides to incorporate the negative margins */
  --margin-top-override: calc(var(--margin-top) - var(--card-padding));   /* = -12px */
  --margin-right-override: calc(var(--margin-right) - var(--card-padding));   /* = -12px */
  --margin-bottom-override: calc(var(--margin-bottom) - var(--card-padding));   /* = -12px */
  --margin-left-override: calc(var(--margin-left) - var(--card-padding));   /* = -12px */

  margin-top: var(--margin-top-override);   /* = -12px */
  margin-right: var(--margin-right-override);   /* = -12px */
  margin-bottom: var(--margin-bottom-override);   /* = -12px */
  margin-left: var(--margin-left-override);   /* = -12px */

  /* Reset the overrides on direct children */
  :where(&) > * {
    --margin-top-override: initial;
    --margin-right-override: initial;
    --margin-bottom-override: initial;
    --margin-left-override: initial;
  }
  &:where(:any-link, button, label) {
    @media (hover: hover) {
      &:where(:hover) {
        background-color: var(--gray-a3);   /* = var(--mauve-a3) */
        &:where(:focus-visible) {
          background-color: var(--focus-a2);   /* = var(--amber-a2) */
        }
      }
    }
    &:where([data-state='open']) {
      background-color: var(--gray-a3);   /* = var(--mauve-a3) */
      &:where(:focus-visible) {
        background-color: var(--focus-a2);   /* = var(--amber-a2) */
      }
    }
    &:where(:active:not([data-state='open'])) {
      background-color: var(--gray-a4);   /* = var(--mauve-a4) */
      &:where(:focus-visible) {
        background-color: var(--focus-a2);   /* = var(--amber-a2) */
      }
    }
  }
}
